/*cria o database*/
create database db_mercearia_lino;

/*ativa o database criado para ser utilizado*/
USE db_mercearia_lino;

CREATE TABLE fornecedor (
	id INT AUTO_INCREMENT NOT NULL PRIMARY KEY,
	nome TEXT NOT NULL,
    telefone TEXT,
    site TEXT
);

CREATE TABLE produto (
	id INT AUTO_INCREMENT NOT NULL PRIMARY KEY,
    nome TEXT NOT NULL,
    fk_id_fornecedor INT NOT NULL,
    precoUnitario double NOT NULL,
    categoriaProduto TEXT NOT NULL,
    quantidadeEstoque INTEGER NOT NULL,
    unidadeMedida TEXT NOT NULL,
    ativo TINYINT(1) DEFAULT 1,
    
    FOREIGN KEY (fk_id_fornecedor) REFERENCES fornecedor(id)
);

CREATE TABLE venda (
	id INT AUTO_INCREMENT NOT NULL PRIMARY KEY,
	fk_id_produto INT NOT NULL,
	quantidade INT NOT NULL,
    formaPagemento TEXT NOT NULL,
    data_venda DATE,
    
	FOREIGN KEY (fk_id_produto) REFERENCES produto(id)
);
